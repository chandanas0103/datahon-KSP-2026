   213→   252→   252→   252→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
   214→   253→   253→   253→  const [count, setCount] = useState(0)
   215→   254→   254→   254→  const ref = useRef<HTMLSpanElement>(null)
   216→   255→   255→   255→  const hasAnimated = useRef(false)
   217→   256→   256→   256→  useEffect(() => {
   218→   257→   257→   257→    if (hasAnimated.current) return
   219→   258→   258→   258→    hasAnimated.current = true
   220→   259→   259→   259→    let start = 0
   221→   260→   260→   260→    const step = (timestamp: number) => {
   222→   261→   261→   261→      if (!start) start = timestamp
   223→   262→   262→   262→      const progress = Math.min((timestamp - start) / duration, 1)
   224→   263→   263→   263→      const eased = 1 - Math.pow(1 - progress, 3)
   225→   264→   264→   264→      setCount(Math.floor(eased * target))
   226→   265→   265→   265→      if (progress < 1) requestAnimationFrame(step)
   227→   266→   266→   266→    }
   228→   267→   267→   267→    requestAnimationFrame(step)
   229→   268→   268→   268→  }, [target, duration])
   230→   269→   269→   269→  return <span ref={ref}>{count.toLocaleString()}</span>
   231→   270→   270→   270→}
