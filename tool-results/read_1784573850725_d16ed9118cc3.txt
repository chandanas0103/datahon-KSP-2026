   252→   252→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
   253→   253→  const [count, setCount] = useState(0)
   254→   254→  const ref = useRef<HTMLSpanElement>(null)
   255→   255→  const hasAnimated = useRef(false)
   256→   256→  useEffect(() => {
   257→   257→    if (hasAnimated.current) return
   258→   258→    hasAnimated.current = true
   259→   259→    let start = 0
   260→   260→    const step = (timestamp: number) => {
   261→   261→      if (!start) start = timestamp
   262→   262→      const progress = Math.min((timestamp - start) / duration, 1)
   263→   263→      const eased = 1 - Math.pow(1 - progress, 3)
   264→   264→      setCount(Math.floor(eased * target))
   265→   265→      if (progress < 1) requestAnimationFrame(step)
   266→   266→    }
   267→   267→    requestAnimationFrame(step)
   268→   268→  }, [target, duration])
   269→   269→  return <span ref={ref}>{count.toLocaleString()}</span>
   270→   270→}

   301→   301→export default function Home() {
   302→   302→  const [messages, setMessages] = useState<Message[]>([])
   303→   303→  const [input, setInput] = useState('')
   304→   304→  const [isSending, setIsSending] = useState(false)
   305→   305→  const [isListening, setIsListening] = useState(false)
   306→   306→  const [stats, setStats] = useState<Stats | null>(null)
   307→   307→  const [history, setHistory] = useState<HistoryItem[]>([])
   308→   308→  const [sidebarOpen, setSidebarOpen] = useState(false)
   309→   309→  const scrollRef = useRef<HTMLDivElement>(null)
   310→   310→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   311→   311→  const recognitionRef = useRef<SpeechRecognition | null>(null)
   312→   312→
   313→   313→  useEffect(() => { fetch('/api/stats').then(r => r.json()).then(setStats).catch(() => {}) }, [])
   314→   314→  useEffect(() => { fetch('/api/history').then(r => r.json()).then(setHistory).catch(() => {}) }, [])
   315→   315→  useEffect(() => { if (scrollRef.current) scrollRef.current.scrollTop = scrollRef.current.scrollHeight }, [messages])
   316→   316→
   317→   317→  // Refresh history after each new message
   318→   318→  useEffect(() => {
   319→   319→    if (messages.length > 0 && messages[messages.length - 1]?.role === 'assistant' && !messages[messages.length - 1]?.isLoading) {
   320→   320→      fetch('/api/history').then(r => r.json()).then(setHistory).catch(() => {})
   321→   321→    }
   322→   322→  }, [messages.length])
   323→   323→
   324→   324→  const sendMessage = useCallback(async (question: string) => {
   325→   325→    if (!question.trim() || isSending) return
   326→   326→    const userMsg: Message = { id: crypto.randomUUID(), role: 'user', content: question.trim() }
   327→   327→    const loadingMsg: Message = { id: crypto.randomUUID(), role: 'assistant', content: '', isLoading: true }
   328→   328→    setMessages((prev) => [...prev, userMsg, loadingMsg])
   329→   329→    setInput('')
   330→   330→    setIsSending(true)
   331→   331→    try {
   332→   332→      const res = await fetch('/api/chat', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ question: question.trim() }) })
   333→   333→      const data = await res.json()
   334→   334→      const assistantMsg: Message = {
   335→   335→        id: crypto.randomUUID(), role: 'assistant',
   336→   336→        content: data.answer || 'No response generated.',
   337→   337→        sql: data.sql || null, results: data.results || [],
   338→   338→        error: data.error, confidence: data.confidence || 'medium',
   339→   339→        translatedQuestion: data.translatedQuestion || null,
   340→   340→        responseTime: data.responseTime || 0,
   341→   341→        followups: data.followups || [],
   342→   342→      }
   343→   343→      setMessages((prev) => [...prev.slice(0, -1), assistantMsg])
   344→   344→    } catch {
   345→   345→      setMessages((prev) => [...prev.slice(0, -1), { id: crypto.randomUUID(), role: 'assistant', content: 'Failed to connect. Please check your connection and try again.', error: 'Network error' }])
   346→   346→    } finally { setIsSending(false) }
   347→   347→  }, [isSending])
   348→   348→
   349→   349→  const handleKeyDown = useCallback((e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   350→   350→    if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); sendMessage(input) }
   351→   351→  }, [input, sendMessage])
   352→   352→
   353→   353→  // Voice input
   354→   354→  const toggleVoice = useCallback(() => {
   355→   355→    if (typeof window === 'undefined' || !('SpeechRecognition' in window || 'webkitSpeechRecognition' in window)) return
   356→   356→    if (isListening) {
   357→   357→      recognitionRef.current?.stop()
   358→   358→      setIsListening(false)
   359→   359→      return
   360→   360→    }
   361→   361→    const SR = window.SpeechRecognition || window.webkitSpeechRecognition
   362→   362→    const recognition = new SR()
   363→   363→    recognition.continuous = false
   364→   364→    recognition.interimResults = false
   365→   365→    recognition.lang = 'kn-IN'
   366→   366→    recognition.onresult = (event: SpeechRecognitionEvent) => {
   367→   367→      const transcript = event.results[0][0].transcript
   368→   368→      setInput(transcript)
   369→   369→      setIsListening(false)
   370→   370→    }
   371→   371→    recognition.onerror = () => setIsListening(false)
