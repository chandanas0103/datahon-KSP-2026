   177→function ChartPanel({ results }: { results: Record<string, unknown>[] }) {
   178→  if (!results || results.length === 0) return null
   179→  const columns = Object.keys(results[0])
   180→  const countCol = columns.find((c) => c.toLowerCase().includes('count') || c.toLowerCase().includes('total') || c.toLowerCase() === 'case_count' || c.toLowerCase() === 'total_cases')
   181→  const labelCol = columns.find((c) => c !== countCol && typeof results[0][c] === 'string')
   182→  const isPieSuitable = labelCol && countCol && (
   183→    labelCol.toLowerCase().includes('status') ||
   184→    labelCol.toLowerCase().includes('category') ||
   185→    labelCol.toLowerCase().includes('gender') ||
   186→    labelCol.toLowerCase().includes('priority') ||
   187→    results.length <= 8
   188→  )
   189→  const chartData = results.map((row) => ({
   190→    name: String(row[labelCol || columns[0]] ?? ''),
   191→    value: Number(row[countCol || columns[1]] ?? 0),
   192→  }))
   193→  if (!countCol || !labelCol) return null
   194→  if (isPieSuitable) {
   195→    return (
   196→      <div className="mt-3 h-64 w-full">
   197→        <ResponsiveContainer width="100%" height="100%">
   198→          <PieChart>
   199→            <Pie data={chartData} cx="50%" cy="50%" innerRadius={50} outerRadius={90} paddingAngle={2} dataKey="value"
   200→              label={({ name, percent }) => `${name} (${(percent * 100).toFixed(0)}%)`} labelLine>
   201→              {chartData.map((_, index) => (<Cell key={`cell-${index}`} fill={CHART_COLORS[index % CHART_COLORS.length]} />))}
   202→            </Pie>
   203→            <Legend />
   204→            <RechartsTooltip contentStyle={{ backgroundColor: 'hsl(var(--popover))', border: '1px solid hsl(var(--border))', borderRadius: '8px', fontSize: '12px' }} />
   205→          </PieChart>
   206→        </ResponsiveContainer>
   207→      </div>
   208→    )
   209→  }
   210→  return (
   211→    <div className="mt-3 h-64 w-full">
   212→      <ResponsiveContainer width="100%" height="100%">
   213→        <BarChart data={chartData} margin={{ top: 5, right: 20, bottom: 5, left: 0 }}>
   214→          <CartesianGrid strokeDasharray="3 3" className="stroke-border" />

   225→function SqlBlock({ sql, explanation, timing, onToggleExplanation, showExplanation }: {
   226→  sql: string

   229→  onToggleExplanation?: () => void

   231→}) {

   233→  const [expanded, setExpanded] = useState(false)

   235→  return (
   236→    <div className="mt-3 rounded-lg border border-border/50 bg-muted/30 overflow-hidden">
   237→      <button onClick={() => setExpanded(!expanded)} className="w-full flex items-center justify-between px-3 py-2 text-xs font-medium text-muted-foreground hover:bg-muted/50 transition-colors">
   238→        <span className="flex items-center gap-1.5"><Database className="h-3.5 w-3.5" />Generated SQL Query</span>
   239→        {expanded ? <ChevronUp className="h-3.5 w-3.5" /> : <ChevronDown className="h-3.5 w-3.5" />}

   241→      {expanded && (
   242→        <div className="relative">
   243→          <pre className="px-3 py-2 text-xs font-mono overflow-x-auto text-foreground/80 leading-relaxed">{sql}</pre>
   244→          <div className="flex items-center gap-1 px-3 pb-2">
   245→            <Button variant="ghost" size="sm" className="h-6 px-2 text-[10px] gap-1" onClick={handleCopy}>
   246→              {copied ? <Check className="h-3 w-3 text-green-500" /> : <><Copy className="h-3 w-3" />Copy</>}
   247→            </Button>
   248→            {explanation && onToggleExplanation && (

   264→              <p className="text-[10px] font-medium text-muted-foreground mb-1.5 flex items-center gap-1"><Timer className="h-2.5 w-2.5" />Response Timing Breakdown</p>
   265→              <div className="flex flex-wrap gap-3">
   266→                {timing.translation != null && (
   267→                  <span className="text-[10px] text-muted-foreground">Translation: <span className="text-foreground/70 font-mono">{(timing.translation / 1000).toFixed(1)}s</span></span>
   268→                )}
   269→                {timing.sqlGeneration != null && (
   270→                  <span className="text-[10px] text-muted-foreground">SQL Gen: <span className="text-foreground/70 font-mono">{(timing.sqlGeneration / 1000).toFixed(1)}s</span></span>
   271→                )}
   272→                {timing.confidence != null && (
   273→                  <span className="text-[10px] text-muted-foreground">Confidence: <span className="text-foreground/70 font-mono">{(timing.confidence / 1000).toFixed(1)}s</span></span>
   274→                )}
   275→              </div>
   276→            </div>

   278→        </div>
   279→      )}

   362→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
   363→  const [count, setCount] = useState(0)
   364→  const hasAnimated = useRef(false)
   365→  useEffect(() => {
   366→    if (hasAnimated.current) return
   367→    hasAnimated.current = true
   368→    let start = 0
   369→    const step = (timestamp: number) => {
   370→      if (!start) start = timestamp
   371→      const progress = Math.min((timestamp - start) / duration, 1)
   372→      const eased = 1 - Math.pow(1 - progress, 3)
   373→      setCount(Math.floor(eased * target))
   374→      if (progress < 1) requestAnimationFrame(step)
   375→    }
   376→    requestAnimationFrame(step)
   377→  }, [target, duration])
   378→  return <span>{count.toLocaleString()}</span>
   379→}

   408→function HistorySidebar({ history, onQuery, isOpen, onToggle }: { history: HistoryItem[]; onQuery: (q: string) => void; isOpen: boolean; onToggle: () => void }) {
   409→  const timeAgo = (dateStr: string) => {
   410→    const now = Date.now()
   411→    const then = new Date(dateStr).getTime()
   412→    const diff = now - then
   413→    if (diff < 60000) return 'just now'
   414→    if (diff < 3600000) return `${Math.floor(diff / 60000)}m ago`
   415→    if (diff < 86400000) return `${Math.floor(diff / 3600000)}h ago`
   416→    return `${Math.floor(diff / 86400000)}d ago`
   417→  }
   418→
   419→  return (
   420→    <>
   421→      {isOpen && (
   422→        <div className="fixed inset-0 bg-black/50 z-40 lg:hidden" onClick={onToggle} />
   423→      )}
   424→      <aside className={`fixed lg:relative z-50 lg:z-auto top-0 left-0 h-full w-72 flex-shrink-0 border-r bg-card/95 backdrop-blur-lg transition-transform duration-300 ease-in-out ${isOpen ? 'translate-x-0' : '-translate-x-full'}`}>
   425→        <div className="flex flex-col h-full">
   426→          <div className="p-4 border-b flex items-center justify-between">
   427→            <h2 className="text-xs font-semibold text-muted-foreground uppercase tracking-wider flex items-center gap-1.5">

   475→// ─── Typing Animation Component ───────────────────────────
   476→
   477→function TypingHero({ onComplete }: { onComplete?: () => void }) {
   478→  const [phraseIdx, setPhraseIdx] = useState(0)
   479→  const [charIdx, setCharIdx] = useState(0)
   480→  const [isDeleting, setIsDeleting] = useState(false)
   481→  const [display, setDisplay] = useState('')
   482→
   483→  useEffect(() => {
   484→    const currentPhrase = TYPING_PHRASES[phraseIdx]
   485→    let timeout: ReturnType<typeof setTimeout>
   486→
   487→    if (!isDeleting && charIdx < currentPhrase.length) {
   488→      timeout = setTimeout(() => {
   489→        setDisplay(currentPhrase.slice(0, charIdx + 1))
   490→        setCharIdx(charIdx + 1)
   491→      }, 50 + Math.random() * 40)
   492→    } else if (!isDeleting && charIdx === currentPhrase.length) {
   493→      timeout = setTimeout(() => setIsDeleting(true), 2000)
   494→    } else if (isDeleting && charIdx > 0) {
   495→      timeout = setTimeout(() => {
   496→        setDisplay(currentPhrase.slice(0, charIdx - 1))
   497→        setCharIdx(charIdx - 1)
   498→      }, 25)
   499→    } else if (isDeleting && charIdx === 0) {
   500→      setIsDeleting(false)
   501→      const nextIdx = (phraseIdx + 1) % TYPING_PHRASES.length
   502→      setPhraseIdx(nextIdx)
   503→    }
   504→
   505→    return () => clearTimeout(timeout)
   506→  }, [charIdx, isDeleting, phraseIdx])
   507→
   508→  return (
   509→    <div className="h-8 flex items-center justify-center">
   510→      <span className="text-base text-primary/90 font-mono">
   511→        {display}
   512→        <span className="animate-pulse">|</span>
   513→      </span>
   514→    </div>
   515→  )
   516→}
