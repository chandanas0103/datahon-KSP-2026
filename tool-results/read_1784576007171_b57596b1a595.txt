   158→function ChartPanel({ results }: { results: Record<string, unknown>[] }) {
   159→  if (!results || results.length === 0) return null
   160→  const columns = Object.keys(results[0])
   161→  const countCol = columns.find((c) => c.toLowerCase().includes('count') || c.toLowerCase().includes('total') || c.toLowerCase() === 'case_count' || c.toLowerCase() === 'total_cases')
   162→  const labelCol = columns.find((c) => c !== countCol && typeof results[0][c] === 'string')
   163→  const isPieSuitable = labelCol && countCol && (
   164→    labelCol.toLowerCase().includes('status') ||
   165→    labelCol.toLowerCase().includes('category') ||
   166→    labelCol.toLowerCase().includes('gender') ||
   167→    labelCol.toLowerCase().includes('priority') ||
   168→    results.length <= 8
   169→  )
   170→  const chartData = results.map((row) => ({
   171→    name: String(row[labelCol || columns[0]] ?? ''),
   172→    value: Number(row[countCol || columns[1]] ?? 0),
   173→  }))
   174→  if (!countCol || !labelCol) return null
   175→  if (isPieSuitable) {
   176→    return (
   177→      <div className="mt-3 h-64 w-full">
   178→        <ResponsiveContainer width="100%" height="100%">
   179→          <PieChart>
   180→            <Pie data={chartData} cx="50%" cy="50%" innerRadius={50} outerRadius={90} paddingAngle={2} dataKey="value"
   181→              label={({ name, percent }) => `${name} (${(percent * 100).toFixed(0)}%)`} labelLine>
   182→              {chartData.map((_, index) => (<Cell key={`cell-${index}`} fill={CHART_COLORS[index % CHART_COLORS.length]} />))}
   183→            </Pie>
   184→            <Legend />
   185→            <RechartsTooltip contentStyle={{ backgroundColor: 'hsl(var(--popover))', border: '1px solid hsl(var(--border))', borderRadius: '8px', fontSize: '12px' }} />
   186→          </PieChart>
   187→        </ResponsiveContainer>
   188→      </div>
   189→    )
   190→  }
   191→  return (
   192→    <div className="mt-3 h-64 w-full">
   193→      <ResponsiveContainer width="100%" height="100%">
   194→        <BarChart data={chartData} margin={{ top: 5, right: 20, bottom: 5, left: 0 }}>
   195→          <CartesianGrid strokeDasharray="3 3" className="stroke-border" />
   196→          <XAxis dataKey="name" tick={{ fontSize: 11 }} angle={-30} textAnchor="end" height={60} />

   304→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
   305→  const [count, setCount] = useState(0)
   306→  const hasAnimated = useRef(false)
   307→  useEffect(() => {
   308→    if (hasAnimated.current) return
   309→    hasAnimated.current = true
   310→    let start = 0
   311→    const step = (timestamp: number) => {
   312→      if (!start) start = timestamp
   313→      const progress = Math.min((timestamp - start) / duration, 1)
   314→      const eased = 1 - Math.pow(1 - progress, 3)
   315→      setCount(Math.floor(eased * target))
   316→      if (progress < 1) requestAnimationFrame(step)
   317→    }
   318→    requestAnimationFrame(step)
   319→  }, [target, duration])
   320→  return <span>{count.toLocaleString()}</span>
   321→}

   350→function HistorySidebar({ history, onQuery, isOpen, onToggle }: { history: HistoryItem[]; onQuery: (q: string) => void; isOpen: boolean; onToggle: () => void }) {
   351→  // Format relative time
   352→  const timeAgo = (dateStr: string) => {
   353→    const now = Date.now()
   354→    const then = new Date(dateStr).getTime()
   355→    const diff = now - then
   356→    if (diff < 60000) return 'just now'
   357→    if (diff < 3600000) return `${Math.floor(diff / 60000)}m ago`
   358→    if (diff < 86400000) return `${Math.floor(diff / 3600000)}h ago`
   359→    return `${Math.floor(diff / 86400000)}d ago`
   360→  }
   361→
   362→  return (
   363→    <>
   364→      {/* Mobile overlay */}
   365→      {isOpen && (
   366→        <div className="fixed inset-0 bg-black/50 z-40 lg:hidden" onClick={onToggle} />
   367→      )}
   368→      {/* Sidebar */}
   369→      <aside className={`fixed lg:relative z-50 lg:z-auto top-0 left-0 h-full w-72 flex-shrink-0 border-r bg-card/95 backdrop-blur-lg transition-transform duration-300 ease-in-out lg:translate-x-0 ${isOpen ? 'translate-x-0' : '-translate-x-full'}`}>
   370→        <div className="flex flex-col h-full">
   371→          <div className="p-4 border-b flex items-center justify-between">
   372→            <h2 className="text-xs font-semibold text-muted-foreground uppercase tracking-wider flex items-center gap-1.5">
   373→              <History className="h-3.5 w-3.5" />Query History
   374→            </h2>
   375→            <Button variant="ghost" size="icon" className="h-7 w-7 lg:hidden" onClick={onToggle}>
   376→              <PanelLeftClose className="h-4 w-4" />
   377→            </Button>
   378→          </div>
   379→          <ScrollArea className="flex-1">
   380→            <div className="p-2 space-y-0.5">
   381→              {history.length === 0 ? (
   382→                <div className="flex flex-col items-center justify-center py-12 text-center">
   383→                  <MessageSquare className="h-8 w-8 text-muted-foreground/20 mb-3" />
   384→                  <p className="text-xs text-muted-foreground/50">No queries yet</p>
   385→                  <p className="text-[10px] text-muted-foreground/30 mt-1">Your past queries will appear here</p>
   386→                </div>
   387→              ) : (
   388→                history.map((item) => (
   389→                  <button key={item.id} onClick={() => { onQuery(item.question); if (window.innerWidth < 1024) onToggle() }} className="w-full text-left px-3 py-2.5 rounded-lg hover:bg-muted/50 transition-colors group">
   390→                    <p className="text-xs text-foreground/80 group-hover:text-foreground line-clamp-2 leading-relaxed">{item.question}</p>
   391→                    <p className="text-[10px] text-muted-foreground/40 mt-1 flex items-center gap-1">
   392→                      <Clock className="h-2.5 w-2.5" />{timeAgo(item.createdAt)}
   393→                    </p>
   394→                  </button>
   395→                ))
   396→              )}
   397→            </div>
   398→          </ScrollArea>
   399→          {history.length > 0 && (
   400→            <div className="p-3 border-t">
   401→              <p className="text-[10px] text-muted-foreground/40 text-center">{history.length} recent {history.length === 1 ? 'query' : 'queries'}</p>
   402→            </div>
   403→          )}
   404→        </div>
   405→      </aside>
   406→    </>
   407→  )
   408→}

   439→  // Refresh history after each new assistant message
   440→  const lastMsg = messages[messages.length - 1]

   447→  const sendMessage = useCallback(async (question: string) => {
   448→    if (!question.trim() || isSending) return

   450→    const loadingMsg: Message = { id: crypto.randomUUID(), role: 'assistant', content: '', isLoading: true }

   452→    setInput('')

   454→    try {
   455→      const res = await fetch('/api/chat', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ question: question.trim() }) })
   456→      const data = await res.json()
   457→      const assistantMsg: Message = {
   458→        id: crypto.randomUUID(),
   459→        role: 'assistant',
   460→        content: data.answer || 'No response generated.',
   461→        sql: data.sql || null,
   462→        results: data.results || [],
   463→        error: data.error,
   464→        confidence: data.confidence || 'medium',
   465→        translatedQuestion: data.translatedQuestion || null,
   466→        responseTime: data.responseTime || 0,
   467→        followups: data.followups || [],
   468→        selfHealed: data.selfHealed || false,
   469→        retryCount: data.retryCount || 0,
   470→      }
   471→      setMessages((prev) => [...prev.slice(0, -1), assistantMsg])
   472→    } catch {
   473→      setMessages((prev) => [...prev.slice(0, -1), {
   474→        id: crypto.randomUUID(), role: 'assistant',
   475→        content: 'Failed to connect to the server. Please check your connection and try again.',
   476→        error: 'Network error',
   477→      }])
   478→    } finally { setIsSending(false) }
   479→  }, [isSending])
   480→
   481→  const handleKeyDown = useCallback((e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   482→    if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); sendMessage(input) }
   483→  }, [input, sendMessage])
   484→
   485→  // Voice input with language cycling
   486→  const toggleVoice = useCallback(() => {
   487→    if (typeof window === 'undefined' || !('SpeechRecognition' in window || 'webkitSpeechRecognition' in window)) return
   488→    if (isListening) {
   489→      recognitionRef.current?.stop()
   490→      setIsListening(false)
   491→      return
   492→    }
   493→    const SR = (window as unknown as { SpeechRecognition: typeof SpeechRecognition; webkitSpeechRecognition: typeof SpeechRecognition }).SpeechRecognition || (window as unknown as { webkitSpeechRecognition: typeof SpeechRecognition }).webkitSpeechRecognition
   494→    const recognition = new SR()
   495→    recognition.continuous = false
   496→    recognition.interimResults = false
   497→    recognition.lang = voiceLang
   498→    recognition.onresult = (event: SpeechRecognitionEvent) => {
   499→      const transcript = event.results[0][0].transcript
   500→      setInput(transcript)
   501→      setIsListening(false)
   502→    }
   503→    recognition.onerror = () => setIsListening(false)
   504→    recognition.onend = () => setIsListening(false)
   505→    recognitionRef.current = recognition
   506→    recognition.start()
   507→    setIsListening(true)
   508→  }, [isListening, voiceLang])
   509→
   510→  const cycleVoiceLang = useCallback(() => {
   511→    const langs: Array<'kn-IN' | 'hi-IN' | 'en-IN'> = ['kn-IN', 'hi-IN', 'en-IN']
   512→    const labels = { 'kn-IN': 'KN', 'hi-IN': 'HI', 'en-IN': 'EN' }
   513→    const idx = langs.indexOf(voiceLang)
   514→    const next = langs[(idx + 1) % langs.length]
   515→    setVoiceLang(next)
   516→  }, [voiceLang])
   517→
   518→  // PDF Export
   519→  const handleExport = useCallback((msg: Message, question: string) => {

   521→      method: 'POST',
   522→      headers: { 'Content-Type': 'application/json' },
   523→      body: JSON.stringify({
   524→        question,
   525→        answer: msg.content,
   526→        sql: msg.sql,
   527→        results: msg.results,
   528→        confidence: msg.confidence,
   529→        responseTime: msg.responseTime,
   530→      }),
   531→    })
   532→    .then(res => res.blob())
   533→    .then(blob => {
   534→      const url = URL.createObjectURL(blob)
   535→      const a = document.createElement('a')
   536→      a.href = url
   537→      a.download = `ksp-crime-report-${new Date().toISOString().slice(0, 10)}.html`
   538→      document.body.appendChild(a)
   539→      a.click()
   540→      document.body.removeChild(a)
   541→      URL.revokeObjectURL(url)
   542→    })
   543→    .catch(() => {
   544→      // Fallback to print
   545→      window.print()
   546→    })
   547→  }, [])
   548→
   549→  const clearChat = useCallback(() => { setMessages([]) }, [])
   550→
   551→  const voiceLangLabel = { 'kn-IN': 'KN', 'hi-IN': 'HI', 'en-IN': 'EN' }[voiceLang]
   552→
   553→  return (

   560→          isOpen={sidebarOpen}
   561→          onToggle={() => setSidebarOpen(!sidebarOpen)}
   562→        />
   563→
   564→        {/* Main Content */}
   565→        <div className="flex-1 flex flex-col min-w-0">
   566→          {/* Header */}
   567→          <header className="no-print flex-shrink-0 border-b bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/60 z-30">
   568→            <div className="max-w-4xl mx-auto px-4 sm:px-6 h-14 flex items-center justify-between">
   569→              <div className="flex items-center gap-3">
   570→                <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => setSidebarOpen(!sidebarOpen)}>
   571→                  {sidebarOpen ? <PanelLeftClose className="h-4 w-4" /> : <PanelLeft className="h-4 w-4" />}
   572→                </Button>
   573→                <div className="w-8 h-8 rounded-lg bg-primary flex items-center justify-center shadow-md shadow-primary/20">
   574→                  <Shield className="h-4.5 w-4.5 text-primary-foreground" />
   575→                </div>
   576→                <div>
   577→                  <h1 className="text-sm font-semibold leading-tight">KSP Crime Intelligence</h1>
   578→                  <p className="text-[10px] text-muted-foreground">Conversational AI for Crime Database</p>
   579→                </div>
   580→              </div>
   581→              <div className="flex items-center gap-2">
   582→                <Tooltip>
   583→                  <TooltipTrigger asChild>
   584→                    <a href="/about" className="text-xs text-muted-foreground hover:text-foreground transition-colors flex items-center gap-1 px-2 py-1 rounded-md hover:bg-muted/50">
   585→                      <FileText className="h-3 w-3" />About
   586→                    </a>
   587→                  </TooltipTrigger>
   588→                  <TooltipContent>Architecture & features</TooltipContent>
   589→                </Tooltip>
   590→                {messages.length > 0 && (
   591→                  <Tooltip>
   592→                    <TooltipTrigger asChild>
   593→                      <Button variant="ghost" size="sm" className="h-7 text-xs text-muted-foreground gap-1" onClick={clearChat}>
   594→                        <X className="h-3 w-3" />Clear
   595→                      </Button>
   596→                    </TooltipTrigger>
   597→                    <TooltipContent>Clear conversation</TooltipContent>
   598→                  </Tooltip>
   599→                )}
   600→              </div>
   601→            </div>
   602→          </header>
   603→
   604→          {/* Messages Area */}
   605→          <div ref={scrollRef} className="flex-1 overflow-y-auto px-4 sm:px-6 py-6">
   606→            <div className="max-w-3xl mx-auto">
   607→              {messages.length === 0 ? (
   608→                <div className="flex flex-col items-center justify-center min-h-[65vh] text-center space-y-6">
   609→                  <div className="relative">
   610→                    <div className="w-16 h-16 rounded-2xl bg-primary/10 flex items-center justify-center">
   611→                      <MessageSquare className="h-8 w-8 text-primary" />
   612→                    </div>

   614→                      <Zap className="h-3 w-3 text-white" />
   615→                    </div>
   616→                  </div>
   617→                  <div className="space-y-2">
   618→                    <h2 className="text-2xl font-bold tracking-tight">Ask anything about crime data</h2>
   619→                    <p className="text-sm text-muted-foreground max-w-lg mx-auto">
   620→                      Query the Karnataka State Police crime database using natural language.
   621→                      Get instant answers with charts, confidence scores, and exportable reports.
   622→                    </p>
   623→                  </div>
   624→                  <div className="flex flex-wrap justify-center gap-2">
   625→                    <FeaturePill icon={ShieldCheck} label="Self-healing SQL" />
   626→                    <FeaturePill icon={Languages} label="Multilingual (EN/KN/HI)" />
   627→                    <FeaturePill icon={ShieldQuestion} label="Confidence Scoring" />
   628→                    <FeaturePill icon={Mic} label="Voice Input" />
   629→                    <FeaturePill icon={FileText} label="Report Export" />
   630→                    <FeaturePill icon={Sparkles} label="Smart Follow-ups" />
   631→                  </div>
   632→                  {stats && <DashboardStats stats={stats} />}
   633→                  <div className="grid grid-cols-1 sm:grid-cols-2 gap-2.5 w-full max-w-2xl mt-2">
   634→                    {SAMPLE_QUESTIONS.slice(0, 6).map((q, i) => (
   635→                      <button key={q} onClick={() => sendMessage(q)} className="text-left text-sm px-4 py-3 rounded-xl border border-border/50 bg-card hover:bg-muted/50 hover:border-border transition-all duration-200 group animate-in fade-in slide-in-from-bottom-2" style={{ animationDelay: `${i * 50}ms`, animationFillMode: 'backwards' }}>
   636→                        <span className="flex items-start gap-2">
   637→                          <Sparkles className="h-4 w-4 text-primary/60 mt-0.5 flex-shrink-0 group-hover:text-primary transition-colors" />
   638→                          <span className="text-foreground/80 group-hover:text-foreground transition-colors">{q}</span>
   639→                        </span>
   640→                      </button>
   641→                    ))}
   642→                  </div>
   643→                </div>
   644→              ) : (
   645→                <div className="space-y-6">
   646→                  {messages.map((msg, idx) => {
   647→                    const userQ = msg.role === 'assistant' && idx > 0 && messages[idx - 1]?.role === 'user'
   648→                      ? messages[idx - 1].content : msg.content
   649→                    return (
   650→                    <MessageBubble
   651→                      key={msg.id}
   652→                      message={msg}
   653→                      userQuestion={userQ}
   654→                      onFollowup={sendMessage}
   655→                      onExport={() => handleExport(msg, userQ)}

   657→                    )
   658→                  })}
   659→                </div>
   660→              )}
   661→            </div>
   662→          </div>
   663→
   664→          {/* Follow-up Quick Bar */}
   665→          {messages.length > 0 && !lastMsg?.isLoading && lastMsg?.followups && lastMsg.followups!.length > 0 && (

   667→              <div className="max-w-3xl mx-auto">
   668→                <ScrollArea className="w-full">
   669→                  <div className="flex gap-1.5 pb-1">
   670→                    {lastMsg.followups!.map((fq) => (
   671→                      <button key={fq} onClick={() => sendMessage(fq)} className="flex-shrink-0 text-xs px-3 py-1.5 rounded-full border border-primary/20 bg-primary/5 text-primary hover:bg-primary/10 transition-colors whitespace-nowrap flex items-center gap-1">
   672→                        <ArrowRight className="h-3 w-3" />{fq}
   673→                      </button>
   674→                    ))}
   675→                  </div>
   676→                </ScrollArea>
   677→              </div>
   678→            </div>
   679→          )}
   680→
   681→          {/* Sample Questions Quick Bar (when in conversation) */}
   682→          {messages.length > 0 && (
   683→            <div className="no-print px-4 sm:px-6 pb-1">
   684→              <div className="max-w-3xl mx-auto">
   685→                <ScrollArea className="w-full">
   686→                  <div className="flex gap-1.5 pb-1">
   687→                    {SAMPLE_QUESTIONS.slice(0, 10).map((q) => (
   688→                      <button key={q} onClick={() => sendMessage(q)} className="flex-shrink-0 text-[11px] px-2.5 py-1 rounded-full border border-border/50 bg-muted/30 hover:bg-muted/60 transition-colors whitespace-nowrap">{q}</button>
   689→                    ))}
   690→                  </div>
   691→                </ScrollArea>
   692→              </div>
   693→            </div>
   694→          )}
   695→
   696→          {/* Input Area */}
   697→          <div className="no-print flex-shrink-0 px-4 sm:px-6 pb-4 pt-2 bg-gradient-to-t from-background via-background to-transparent">
   698→            <div className="max-w-3xl mx-auto">
   699→              <Card className="shadow-lg shadow-primary/5 border-border/50">
   700→                <CardContent className="p-3">
   701→                  <div className="flex gap-2 items-end">
   702→                    <Textarea
   703→                      ref={textareaRef}
   704→                      value={input}
   705→                      onChange={(e) => setInput(e.target.value)}
   706→                      onKeyDown={handleKeyDown}
   707→                      placeholder="Ask in English, Kannada, or Hindi..."
   708→                      className="min-h-[44px] max-h-[120px] resize-none border-0 focus-visible:ring-0 bg-transparent text-sm"
   709→                      rows={1}
   710→                      disabled={isSending}
   711→                    />
   712→                    <div className="flex flex-col gap-1">
   713→                      <Tooltip>
   714→                        <TooltipTrigger asChild>
   715→                          <Button
   716→                            variant={isListening ? 'destructive' : 'ghost'}
   717→                            size="icon"
   718→                            onClick={toggleVoice}
   719→                            disabled={isSending}
   720→                            className="h-10 w-10 rounded-xl relative"
   721→                          >
   722→                            {isListening ? (
   723→                              <><MicOff className="h-4 w-4" /><span className="absolute -top-1 -right-1 w-2.5 h-2.5 bg-red-500 rounded-full animate-ping" /></>) : <Mic className="h-4 w-4" />}
   724→                          </Button>
   725→                        </TooltipTrigger>
   726→                        <TooltipContent>
   727→                          <div className="flex flex-col gap-1">
   728→                            <span>{isListening ? 'Stop listening' : 'Voice input'}</span>
   729→                            <span className="text-[10px] text-muted-foreground">Click mic icon to cycle: KN → HI → EN</span>
   730→                          </div>
   731→                        </TooltipContent>
   732→                      </Tooltip>
   733→                      <Tooltip>
   734→                        <TooltipTrigger asChild>
   735→                          <Button
   736→                            variant="ghost"
   737→                            size="sm"
   738→                            onClick={cycleVoiceLang}
   739→                            disabled={isSending}
   740→                            className="h-5 px-1.5 text-[9px] text-muted-foreground/60 hover:text-foreground"
   741→                          >
   742→                            {voiceLangLabel}
   743→                          </Button>
   744→                        </TooltipTrigger>
   745→                        <TooltipContent>Language: {voiceLang === 'kn-IN' ? 'Kannada' : voiceLang === 'hi-IN' ? 'Hindi' : 'English'}</TooltipContent>
   746→                      </Tooltip>
   747→                    </div>
   748→                    <Button
   749→                      size="icon"
   750→                      onClick={() => sendMessage(input)}
   751→                      disabled={!input.trim() || isSending}
   752→                      className="flex-shrink-0 h-10 w-10 rounded-xl"
   753→                    >
   754→                      {isSending ? <div className="h-4 w-4 animate-spin rounded-full border-2 border-current border-t-transparent" /> : <Send className="h-4 w-4" />}
   755→                    </Button>
   756→                  </div>
   757→                </CardContent>
   758→              </Card>
   759→              <div className="flex items-center justify-center gap-3 mt-2 text-[10px] text-muted-foreground/40">
   760→                <span className="flex items-center gap-1"><Shield className="h-2.5 w-2.5" />KSP Datathon 2026</span>
   761→                <span className="text-border">|</span>
   762→                <span className="flex items-center gap-1"><RefreshCw className="h-2.5 w-2.5" />Self-healing SQL</span>
   763→                <span className="text-border">|</span>
   764→                <span>All queries logged</span>
   765→              </div>
   766→            </div>
   767→          </div>
   768→        </div>
   769→      </div>
   770→    </TooltipProvider>
   771→  )
